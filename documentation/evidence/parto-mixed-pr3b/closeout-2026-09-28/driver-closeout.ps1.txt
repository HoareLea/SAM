function Say($m) { $l = "{0:HH:mm:ss} [{1}] {2}" -f (Get-Date), $Stage, $m; $l; [System.IO.File]::AppendAllText($log, $l + [Environment]::NewLine) }
function Detached([string]$Arguments) { [void](Invoke-CimMethod -ClassName Win32_Process -MethodName Create -Arguments @{ CommandLine = "powershell.exe -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden $Arguments" }) }
function Press([object]$Top, [string]$Id) { Say "  press $Id"; Detached ("-File `"$sp\invoke.ps1`" -ProcessId $ProcessId -Handle {0} -Id {1}" -f $Top.Handle.ToInt64(), $Id) }
function Top([string]$Like, [int]$Seconds) {
    $dl = (Get-Date).AddSeconds($Seconds)
    do { $t = Get-TopWindows $ProcessId | Where-Object { $_.Title -like $Like } | Select-Object -First 1; if ($t) { return $t }; Start-Sleep -Seconds 2 } while ((Get-Date) -lt $dl)
    return $null
}
function BoxText($b) { (Get-Children $b.Handle | ForEach-Object { $t = New-Object System.Text.StringBuilder 4096; [void][W.N]::GetWindowText($_.Handle, $t, 4096); $t.ToString() } | Where-Object { $_ }) -join ' | ' }
function BoxButtons($b) { @(Get-Children $b.Handle | Where-Object { $_.Class -eq 'Button' }) }
function Shot([string]$Name) {
    $b = [System.Windows.Forms.Screen]::PrimaryScreen.Bounds
    $bmp = New-Object System.Drawing.Bitmap $b.Width, $b.Height
    $g = [System.Drawing.Graphics]::FromImage($bmp); $g.CopyFromScreen($b.Location, [System.Drawing.Point]::Empty, $b.Size)
    $p = Join-Path $shots ("{0:HHmmss}-{1}.png" -f (Get-Date), $Name); $bmp.Save($p, [System.Drawing.Imaging.ImageFormat]::Png); $g.Dispose(); $bmp.Dispose()
}
function Txt($root, $Id) { $e = Find-ById $root $Id; if ($e) { $e.Current.Name -replace "`r?`n", ' / ' } else { '<absent>' } }
function Texts($root, $Id) { $e = Find-ById $root $Id; if (-not $e) { return '<absent>' }; (@($e.FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::Text))) | ForEach-Object { $_.Current.Name } | Where-Object { $_ }) -join ' | ') -replace "`r?`n", ' / ' }
function Value($root, $id) { $e = Find-ById $root $id; $v = $null; if ($e -and $e.TryGetCurrentPattern([System.Windows.Automation.ValuePattern]::Pattern, [ref]$v)) { return $v.Current.Value }; return $null }
function Expand($root, $id) { $e = Find-ById $root $id; $p = $null; if ($e -and $e.TryGetCurrentPattern([System.Windows.Automation.ExpandCollapsePattern]::Pattern, [ref]$p)) { $p.Expand(); Start-Sleep -Milliseconds 800; return $true }; return $false }
function DumpNamed($root, [string]$File) {
    $lines = foreach ($e in $root.FindAll($TS::Descendants, [System.Windows.Automation.Condition]::TrueCondition)) {
        try { $c = $e.Current; if ($c.AutomationId -or ($c.Name -and $c.Name -notlike 'SAM.*')) { "{0} id={1} en={2} | {3}" -f $c.ControlType.ProgrammaticName.Replace('ControlType.', ''), $c.AutomationId, $c.IsEnabled, ($c.Name -replace "`r?`n", ' / ') } } catch {}
    }
    $lines | Set-Content -Path (Join-Path $Dir $File) -Encoding UTF8
}
function CopyAll([object]$Top, [string]$File) {
    Set-Clipboard -Value ' '
    Press $Top 'button_CopyAll'
    $dl = (Get-Date).AddSeconds(20); $text = $null
    do { Start-Sleep -Milliseconds 700; $text = Get-Clipboard -Raw } while (((-not $text) -or $text.Trim().Length -eq 0) -and (Get-Date) -lt $dl)
    Set-Content -Path (Join-Path $Dir $File) -Value $text -Encoding UTF8
    Say ("  copied {0} chars to {1}" -f $text.Length, $File)
}
function Fail([string]$m) { Say "FAIL $m"; Shot 'fail'; Set-Content -Path $stop -Value 'stop'; Stop-Process -Id $ProcessId -Force -ErrorAction SilentlyContinue; exit 1 }
# Waits for a window; logs the Part O progress window's text as it changes; dismisses ONLY a one-button info box.
function WaitFor([string]$Like, [int]$Minutes, [string]$What) {
    $dl = (Get-Date).AddMinutes($Minutes); $lastProgress = ''
    while ((Get-Date) -lt $dl) {
        $tops = Get-TopWindows $ProcessId
        $t = $tops | Where-Object { $_.Title -like $Like } | Select-Object -First 1
        if ($t) { return $t }
        $mb = $tops | Where-Object { $_.Class -eq '#32770' } | Select-Object -First 1
        if ($mb) {
            $buttons = BoxButtons $mb; Shot 'messagebox'
            Say ("  MESSAGE BOX while waiting for {0} ({1} button(s)): {2}" -f $What, $buttons.Count, (BoxText $mb))
            if ($buttons.Count -ne 1) { Fail 'a confirmation dialog appeared - not answered by the driver' }
            [void][W.N]::PostMessage($buttons[0].Handle, 0x00F5, [System.IntPtr]::Zero, [System.IntPtr]::Zero); Start-Sleep -Seconds 3
        }
        $pw = $tops | Where-Object { $_.Title -eq 'Part O' -and $_.Class -ne '#32770' } | Select-Object -First 1
        if ($pw) { try { $txt = (@($AE::FromHandle($pw.Handle).FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::Text))) | ForEach-Object { $_.Current.Name } | Where-Object { $_ }) -join ' | ') -replace "`r?`n", ' / '; if ($txt -and $txt -ne $lastProgress) { Say "  progress: $txt"; $lastProgress = $txt } } catch {} }
        if (-not (Get-Process -Id $ProcessId -ErrorAction SilentlyContinue)) { Say 'FAIL the application exited'; exit 3 }
        Start-Sleep -Seconds 3
    }
    return $null
}
