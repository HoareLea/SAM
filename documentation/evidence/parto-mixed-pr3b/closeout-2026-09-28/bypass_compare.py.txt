# PR3B closeout: MG heat-recovery bypass and cooling, DV = true (Nuaire-reply closeout, 24 Sep) vs DV = false
# (PR3B closeout, 28 Sep), from each run's own TAS read-back CSV (GuidanceCoolingResults.ToCsv). Read-only.
import csv, collections, sys

RUNS = [
    ('DV=true  (nuaire-reply closeout 24 Sep)', r'C:\TasOut\nuaire-reply-2026-09-24\03-Iteration3-MG-closeout\000000_SAM_AnalyticalModel-It1a-futureZ1-It3BMG-OperatingAirFlow.csv'),
    ('DV=false (PR3B closeout 28 Sep)', r'C:\TasOut\parto-pr3b-closeout-2026-09-28\02-Iteration3-MG\000000_SAM_AnalyticalModel-It1a-futureZ1-It3BMG-OperatingAirFlow.csv'),
]
TOL = 0.05


def f(x):
    return float(x) if x not in ('', None) else float('nan')


for label, path in RUNS:
    units = collections.defaultdict(list)
    for r in csv.DictReader(open(path)):
        units[r['air_system']].append(r)
    print(label)
    for name, rows in sorted(units.items()):
        c = collections.Counter()
        extract_max = -99.0
        for r in rows:
            to, ta, exl, sup = f(r['intake_C']), f(r['extract_C']), f(r['exchanger_leaving_C']), f(r['supply_C'])
            sen = f(r['dx_sensible_W'])
            extract_max = max(extract_max, ta)
            rule = to >= 12.0 and to < ta and ta >= 19.0
            bypassed = abs(exl - to) <= TOL
            c['hours'] += 1
            c['rule'] += rule
            c['bypassed'] += bypassed
            c['rule_and_bypassed'] += rule and bypassed
            c['recovering'] += (not bypassed)
            c['dx_hours'] += abs(sen) > 1e-6
            c['dx_kWh'] += abs(sen) / 1000.0
        print('  {0}: bypass rule true {1} h, exchanger bypassed {2} h (rule true & bypassed {3}); recovering {4} h; '
              'DX {5} h, {6:.0f} kWh sensible; extract max {7:.1f} C'.format(
                  name, c['rule'], c['bypassed'], c['rule_and_bypassed'], c['recovering'], c['dx_hours'], c['dx_kWh'], extract_max))
