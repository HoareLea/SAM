// PR2F review harness (scratch, not product code).
//   convert <tsd> <samIn> <samOut>        production SAM.Analytical.Tas.Modify.AddResults onto a copy of the model
//   review  <sam> <outDir> [SI|IP] [names] every Space (or the named ones): Design Load PDF + JSON + balance CSV
//   batch   <sam> <outDir> [n]             timing: per-Space context (current UI path) vs one shared context
using PdfSharp.Pdf.IO;
using SAM.Analytical;
using SAM.Core.Reporting;
using SAM.Core.Reporting.Pdf;
using SAM.Units;
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Text;
using AR = SAM.Analytical.Reporting;

static class Program
{
    static readonly CultureInfo IC = CultureInfo.InvariantCulture;

    static int Main(string[] args)
    {
        switch (args[0])
        {
            case "convert": return Convert(args[1], args[2], args[3]);
            case "review": return Review(args[1], args[2], args.Length > 3 ? args[3] : "SI", args.Skip(4).ToArray());
            case "batch": return Batch(args[1], args[2], args.Length > 3 ? int.Parse(args[3]) : int.MaxValue);
            case "scale": return Scale(args[1], args[2], int.Parse(args[3]));
        }
        return 1;
    }

    static AnalyticalModel Load(string sam) => SAM.Core.Convert.ToSAM<AnalyticalModel>(sam).First();

    static int Convert(string tsd, string samIn, string samOut)
    {
        AnalyticalModel model = Load(samIn);
        AdjacencyCluster cluster = model.AdjacencyCluster;
        List<SAM.Core.Result> results = SAM.Analytical.Tas.Modify.AddResults(tsd, cluster);
        Console.WriteLine($"converted results: {results?.Count}");
        model = new AnalyticalModel(model, cluster);
        SAM.Core.Convert.ToFile(new List<SAM.Core.IJSAMObject> { model }, samOut);
        return 0;
    }

    static string F(double v) => v.ToString("0.###", IC);

    static int Review(string sam, string outDir, string units, string[] names)
    {
        Directory.CreateDirectory(outDir);
        AnalyticalModel model = Load(sam);
        UnitStyle unitStyle = units == "IP" ? UnitStyle.Imperial : UnitStyle.SI;

        StringBuilder md = new StringBuilder();
        md.AppendLine($"# {Path.GetFileName(sam)} ({units})");

        foreach (AnalyticalModelParameter p in new[] { AnalyticalModelParameter.HeatingDesignDays, AnalyticalModelParameter.CoolingDesignDays })
        {
            if (model.TryGetValue(p, out SAM.Core.SAMCollection<DesignDay> dds) && dds != null)
                foreach (DesignDay dd in dds)
                    md.AppendLine($"- model {p}: name='{dd.Name}' description='{dd.Description}' date={dd.Year}-{dd.Month}-{dd.Day}");
        }

        List<string> csv = new List<string> { "space,loadType,basis,load_W,sensibleSum_W,residual_W,load+sum_W,displayedLoad,displayedSum,roundingOnly,latentSum_W,designDayName,hourOfDay,hourOfYear,components" };

        List<Space> spaces = model.AdjacencyCluster.GetSpaces();
        if (names.Length != 0) spaces = spaces.Where(x => names.Contains(x.Name)).ToList();

        foreach (Space space in spaces.OrderBy(x => x.Name))
        {
            AR.DocumentContext ctx = AR.Create.DocumentContext(model, new DocumentOptions { UnitSystem = unitStyle });
            Document document = AR.Create.SpaceDesignLoadSummary(ctx, space);
            byte[] pdf = new PdfRenderer().Render(document);
            string safe = string.Concat(space.Name.Select(c => Path.GetInvalidFileNameChars().Contains(c) ? '_' : c));
            File.WriteAllBytes(Path.Combine(outDir, safe + ".pdf"), pdf);
            File.WriteAllText(Path.Combine(outDir, safe + ".json"), document.ToJson());
            int pages;
            using (MemoryStream ms = new MemoryStream(pdf)) pages = PdfReader.Open(ms, PdfDocumentOpenMode.Import).PageCount;

            // Markers actually printed as "—" and where
            List<string> dashes = new List<string>();
            foreach (DocumentSection s in document.Sections)
                foreach (DocumentBlock b in s.Blocks)
                    foreach (FormattedValue fv in BlockValues(b))
                        if (fv.Text == ctx.Formatter.NotAvailableText) dashes.Add($"{s.Id}/{b.Id}:{fv.Note}");

            md.AppendLine();
            md.AppendLine($"## {space.Name} — {pages} page(s), IC='{space.InternalCondition?.Name}'");
            md.AppendLine($"- legend: [{string.Join(" | ", document.Footer.Legend)}]; '—' printed at: {(dashes.Count == 0 ? "none" : string.Join("; ", dashes))}");
            foreach (DocumentSection s in document.Sections)
                foreach (DocumentBlock b in s.Blocks)
                {
                    if (b is NoticeBlock nb) md.AppendLine($"- [{s.Id}] notice {nb.Id}: {nb.Text}");
                    if (b is KeyValueBlock kv && s.Id is "design-criteria" or "heating" or "cooling")
                        foreach (var r in kv.Rows) md.AppendLine($"- [{s.Id}] {r.Label}: {r.Value.Text + (r.Value.Unit == null ? "" : " " + r.Value.Unit)}");
                    if (b is TableBlock tb && (s.Id == "design-criteria"))
                        foreach (var r in tb.Rows) md.AppendLine($"- [{s.Id}] {string.Join(" | ", r.Cells.Select(v => v.Text + (v.Unit == null ? "" : " " + v.Unit)))}");
                }

            foreach (SpaceSimulationResult r in model.AdjacencyCluster.GetResults<SpaceSimulationResult>(space) ?? new List<SpaceSimulationResult>())
            {
                foreach (SpaceSimulationResultParameter pp in new[] { SpaceSimulationResultParameter.DesignDayPeak, SpaceSimulationResultParameter.AnnualPeak })
                {
                    if (!r.TryGetValue(pp, out SpaceLoadPeak peak) || peak == null) continue;
                    var sens = peak.Components.Where(x => x.Key != LoadPeakComponent.OccupancyLatent && x.Key != LoadPeakComponent.EquipmentLatent).ToList();
                    var lat = peak.Components.Where(x => x.Key == LoadPeakComponent.OccupancyLatent || x.Key == LoadPeakComponent.EquipmentLatent).ToList();
                    double sum = sens.Sum(x => x.Value);
                    // heating: load = -sum ; cooling: load = +sum (gain removed)
                    double residual = r.LoadType() == LoadType.Heating ? peak.Load + sum : peak.Load - sum;
                    double displayedSum = sens.Sum(x => Math.Round(x.Value));
                    double displayedLoad = Math.Round(peak.Load);
                    double displayedResidual = r.LoadType() == LoadType.Heating ? displayedLoad + displayedSum : displayedLoad - displayedSum;
                    bool roundingOnly = Math.Abs(residual) < 0.5;
                    csv.Add(string.Join(",", Q(space.Name), r.LoadType(), peak.Basis, F(peak.Load), F(sum), F(residual), F(peak.Load + sum), F(displayedLoad), F(displayedSum), $"{roundingOnly}(displayedResidual={F(displayedResidual)})", F(lat.Sum(x => x.Value)), Q(peak.DesignDayName), peak.HourOfDay?.ToString(), peak.HourOfYear?.ToString(),
                        Q(string.Join("; ", peak.Components.Select(x => $"{x.Key}={F(x.Value)}")))));
                }
            }

            foreach (string d in ctx.Diagnostics.Select(x => x.Text)) md.AppendLine($"- diag: {d}");
        }

        File.WriteAllText(Path.Combine(outDir, "_review.md"), md.ToString());
        File.WriteAllLines(Path.Combine(outDir, "_balance.csv"), csv);
        Console.WriteLine($"{spaces.Count} spaces -> {outDir}");
        return 0;
    }

    static string Q(string s) => "\"" + (s ?? "").Replace("\"", "'") + "\"";

    static IEnumerable<FormattedValue> BlockValues(DocumentBlock b)
    {
        if (b is KeyValueBlock kv) return kv.Rows.Select(x => x.Value);
        if (b is TableBlock tb) return tb.Rows.SelectMany(x => x.Cells);
        return Enumerable.Empty<FormattedValue>();
    }

    static int Scale(string sam, string outDir, int target)
    {
        AnalyticalModel model = Load(sam);
        AdjacencyCluster cluster = model.AdjacencyCluster;
        List<Space> originals = cluster.GetSpaces();
        int copies = target / originals.Count;
        for (int c = 1; c < copies; c++)
            foreach (Space s in originals)
            {
                Space s2 = new Space(Guid.NewGuid(), s, s.Name + "_c" + c, s.Location);
                cluster.AddObject(s2);
                foreach (Panel p in cluster.GetPanels(s) ?? new List<Panel>())
                {
                    Panel p2 = SAM.Analytical.Create.Panel(Guid.NewGuid(), p);
                    cluster.AddObject(p2);
                    cluster.AddRelation(s2, p2);
                }
            }
        model = new AnalyticalModel(model, cluster);
        return BatchModel(model, outDir, int.MaxValue);
    }

    static int Batch(string sam, string outDir, int n) => BatchModel(Load(sam), outDir, n);

    static int BatchModel(AnalyticalModel model, string outDir, int n)
    {
        Directory.CreateDirectory(outDir);
        Stopwatch sw = Stopwatch.StartNew();
        List<Space> spaces = model.AdjacencyCluster.GetSpaces().Take(n).ToList();
        Console.WriteLine($"load {sw.ElapsedMilliseconds} ms, spaces {spaces.Count}, panels {model.AdjacencyCluster.GetPanels().Count}");

        // A: current UI path - a new DocumentContext (cluster deep clone) per Space
        sw.Restart();
        long ctxMs = 0, docMs = 0, pdfMs = 0;
        int take = Math.Min(spaces.Count, 40);
        foreach (Space space in spaces.Take(take))
        {
            Stopwatch s = Stopwatch.StartNew();
            AR.DocumentContext ctx = AR.Create.DocumentContext(model, new DocumentOptions());
            ctxMs += s.ElapsedMilliseconds; s.Restart();
            Document d = AR.Create.SpaceDesignLoadSummary(ctx, space);
            docMs += s.ElapsedMilliseconds; s.Restart();
            byte[] pdf = new PdfRenderer().Render(d);
            pdfMs += s.ElapsedMilliseconds;
        }
        Console.WriteLine($"A per-space context x{take}: total {sw.ElapsedMilliseconds} ms  ctx {ctxMs}  doc {docMs}  pdf {pdfMs}  => {sw.ElapsedMilliseconds / (double)take:0.0} ms/space");

        // B: one shared context (diagnostics then accumulate across Spaces - needs a per-document log in product)
        sw.Restart();
        ctxMs = docMs = pdfMs = 0;
        Stopwatch s0 = Stopwatch.StartNew();
        AR.DocumentContext shared = AR.Create.DocumentContext(model, new DocumentOptions());
        ctxMs = s0.ElapsedMilliseconds;
        long peak = 0;
        long diagnostics = 0;
        foreach (Space space in spaces)
        {
            Stopwatch s = Stopwatch.StartNew();
            // PR2F-1 API: one snapshot, a fresh diagnostics log per document.
            AR.DocumentContext next = shared.WithNewDiagnostics();
            Document d = AR.Create.SpaceDesignLoadSummary(next, space);
            diagnostics += next.Diagnostics.Count();
            docMs += s.ElapsedMilliseconds; s.Restart();
            byte[] pdf = new PdfRenderer().Render(d);
            pdfMs += s.ElapsedMilliseconds;
            File.WriteAllBytes(Path.Combine(outDir, space.Guid + ".pdf"), pdf);
            peak = Math.Max(peak, GC.GetTotalMemory(false));
        }
        Console.WriteLine($"B shared context x{spaces.Count}: total {sw.ElapsedMilliseconds} ms  ctx {ctxMs}  doc {docMs}  pdf {pdfMs}  => {sw.ElapsedMilliseconds / (double)spaces.Count:0.0} ms/space; managed peak {peak / 1048576} MB; per-document diagnostics total {diagnostics}, shared log {shared.Diagnostics.Count()}");
        return 0;
    }
}
