$sp = Split-Path -Parent $MyInvocation.MyCommand.Path
$roots = @('C:\TasOut','C:\Users\Virtual Machine\Documents','C:\Users\Virtual Machine\Nextcloud')
$f = foreach($r in $roots){ Get-ChildItem $r -Recurse -Filter *.tsd -ErrorAction SilentlyContinue }
$u = $f | Where-Object { $_.Length -gt 1000 } | Group-Object { "$($_.Length)|$($_.LastWriteTime.Ticks)" } | ForEach-Object { $_.Group[0] }
$out = "$sp\coolscan2.txt"; Remove-Item $out -ErrorAction SilentlyContinue
Add-Content $out "unique: $($u.Count)"
$n = 0
foreach($x in $u){
  $n++
  Get-Process TSD,CoolScan -ErrorAction SilentlyContinue | Stop-Process -Force -Confirm:$false
  $copy = "$sp\scan_$n.tsd"; Copy-Item $x.FullName $copy -Force
  Add-Content $out "== $($x.FullName)"
  $p = Start-Process -FilePath "$sp\CoolScan.exe" -ArgumentList "`"$copy`"" -RedirectStandardOutput "$sp\one_$n.txt" -NoNewWindow -PassThru
  if (-not $p.WaitForExit(60000)) { $p.Kill(); Add-Content $out "  TIMEOUT" }
  Get-Content "$sp\one_$n.txt" | Add-Content $out
  Remove-Item $copy -ErrorAction SilentlyContinue
}
Get-Process TSD,CoolScan -ErrorAction SilentlyContinue | Stop-Process -Force -Confirm:$false
Add-Content $out "DONE"
