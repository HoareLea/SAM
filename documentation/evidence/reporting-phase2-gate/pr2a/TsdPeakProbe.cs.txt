using System; using TSD; using System.Globalization;
class P {
  static string F(double v){ return v.ToString("0.######", CultureInfo.InvariantCulture); }
  static int[] Sens = {7,8,9,10,11,12,13,14,15,32}; // solar,light,infVent,airMov,BHT,extOpq,extGlz,occS,eqS,AHU
  static void Dump(string tag, ZoneData z, int h, BuildingData bd){
    Console.Write("  "+tag+" h="+h+":");
    foreach (tsdZoneArray a in Enum.GetValues(typeof(tsdZoneArray))) Console.Write(" "+a+"="+F(z.GetHourlyZoneResult(h,(int)a)));
    Console.WriteLine();
    double s=0; foreach(int p in Sens) s+=z.GetHourlyZoneResult(h,p);
    double heat=z.GetHourlyZoneResult(h,5), cool=z.GetHourlyZoneResult(h,6), sensL=z.GetHourlyZoneResult(h,4);
    Console.WriteLine("    sumSensible="+F(s)+" heat="+F(heat)+" cool="+F(cool)+" sensibleLoad="+F(sensL)+"  cool-sum="+F(cool-s)+"  heat+sum="+F(heat+s)+"  (heat-cool)+sum="+F(heat-cool+s));
    if(bd!=null) Console.WriteLine("    ext T="+F(bd.GetHourlyBuildingResult(h,2))+" RH="+F(bd.GetHourlyBuildingResult(h,1)));
  }
  static int Col(Func<int,ZoneData> get, string name){ for(int i=1;i<10000;i++){ var z=get(i); if(z==null) return -1; if(z.name==name) return i; } return -1; }
  static void Main(string[] args){
    var d=new TSDDocument(); if(!d.openReadOnly(args[0])){ Console.WriteLine("open failed"); return; }
    try {
      var sim=d.SimulationData; var bd=sim.GetBuildingData();
      for(int n=1;n<args.Length;n++){
        string zn=args[n]; Console.WriteLine("== "+zn);
        int bi=Col(i=>bd.GetZoneData(i),zn); var bz=bd.GetZoneData(bi);
        foreach(short prm in new short[]{6,5}){
          var a=(object[,])bd.GetPeakZoneGains(new short[]{prm}); float v=(float)a[1,bi-1]; int ix=(int)a[2,bi-1];
          Console.WriteLine(" annual "+(prm==6?"cool":"heat")+" peak="+F(v)+" @"+ix);
          if(ix>0) Dump("annual", bz, ix, bd);
        }
        for(int k=0;k<4;k++){ var c=sim.GetCoolingDesignData(k); if(c==null)break; int ci=Col(i=>c.GetZoneData(i),zn); var cz=c.GetZoneData(ci);
          var a=(object[,])c.GetPeakZoneGains(new short[]{6}); float v=(float)a[1,ci-1]; int ix=(int)a[2,ci-1];
          Console.WriteLine(" CDD["+k+"] "+c.name+" days "+c.firstDay+".."+c.lastDay+" cool peak="+F(v)+" @"+ix); if(ix>0) Dump("CDD", cz, ix, null); }
        for(int k=0;k<4;k++){ var c=sim.GetHeatingDesignData(k); if(c==null)break; int ci=Col(i=>c.GetZoneData(i),zn); var cz=c.GetZoneData(ci);
          var a=(object[,])c.GetPeakZoneGains(new short[]{5}); float v=(float)a[1,ci-1]; int ix=(int)a[2,ci-1];
          Console.WriteLine(" HDD["+k+"] "+c.name+" heat peak="+F(v)+" @"+ix); if(ix>0) Dump("HDD", cz, ix, null); }
      }
    } finally { d.close(); }
  }
}
