using System; using TSD; using System.Globalization;
class P {
  static string F(double v){ return v.ToString("0.###", CultureInfo.InvariantCulture); }
  static void Max(string tag, object arr, Func<int,ZoneData> zone){
    var a=(object[,])arr; int lo=a.GetLowerBound(1), hi=a.GetUpperBound(1); double best=0; int bi=-1, bidx=0;
    for(int c=lo;c<=hi;c++){ double v=(float)a[a.GetLowerBound(0)+1,c]; if(v>best){best=v;bi=c;bidx=(int)a[a.GetLowerBound(0)+2,c];} }
    string n = bi<0? "-" : zone(bi-lo+1).name;
    Console.WriteLine("  {0} maxCool={1} zone={2} idx={3}", tag, F(best), n, bidx);
  }
  static void Main(string[] args){
    var d=new TSDDocument(); if(!d.openReadOnly(args[0])){ Console.WriteLine("  open failed"); return; }
    try {
      var sim=d.SimulationData; var bd=sim.GetBuildingData();
      if(bd!=null) Max("annual", bd.GetPeakZoneGains(new short[]{6}), i=>bd.GetZoneData(i));
      for(int k=0;k<8;k++){ var c=sim.GetCoolingDesignData(k); if(c==null)break; var cc=c; Max("CDD["+k+"] "+c.name, c.GetPeakZoneGains(new short[]{6}), i=>cc.GetZoneData(i)); }
    } finally { d.close(); }
  }
}
